# Windows runtime deps for the Jarvis computer-use engine.
#
# The engine package is installed with `pip install -e ../../computer-use --no-deps`
# (so pip does NOT try to build the Linux-only deps evdev / dbus-fast / pywayland,
# which fail on Windows) — so this file lists the CROSS-PLATFORM runtime deps the
# engine actually needs on Windows, plus the Windows backend's own deps. This keeps
# computer-use/pyproject.toml untouched.

# Cross-platform engine runtime (mirrors the portable subset of pyproject.toml):
mcp>=1.26.0,<2   # 2.x renamed FastMCP -> MCPServer; engine code targets 1.x
fastapi>=0.115
uvicorn[standard]>=0.30
pyyaml>=6.0
pillow>=10.0

# Windows computer-use backend:
pywin32        # Win32: clipboard, windows, Office COM (win32com) + ctypes SendInput helper
mss            # fast multi-monitor screen capture
uiautomation   # UI Automation (ui_* tools); pulls in comtypes
# Built-in Windows OCR (screen_ocr / screen_find_text / click_text) via pywinrt:
winrt-runtime
winrt-Windows.Media.Ocr
winrt-Windows.Graphics.Imaging
winrt-Windows.Storage.Streams
winrt-Windows.Globalization
winrt-Windows.Foundation
winrt-Windows.Foundation.Collections

# Video understanding (computer_use_mcp/video) — keep in sync with the video
# block in computer-use/pyproject.toml:
yt-dlp>=2026.1        # imported as a library; freezes into the exe (no PATH binary)
faster-whisper>=1.2   # default local transcription engine (ctranslate2 + av)
huggingface-hub>=0.25 # model auto-download for faster-whisper
# Cloud backends ship on Windows unconditionally (small pure-python wheels;
# Windows users are less likely to have local whisper CLIs, so the cloud
# fallback matters more here). Selected via GEMINI_API_KEY / OPENAI_API_KEY.
google-genai>=1.0
openai>=1.60
